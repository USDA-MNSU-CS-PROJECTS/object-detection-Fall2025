# Alfalfa Stem Object Detection — pip requirements
#
# Direct dependencies only. Transitive deps (requests, urllib3, cryptography,
# cffi, certifi, PyJWT, attrs, six, etc.) are resolved automatically by pip
# from the packages listed below — pinning them by hand creates resolver
# conflicts on Windows.
#
# Targets Python 3.9 (gradio 3.50.2 is the only Gradio version that works
# reliably with the rest of this stack on Python 3.9).
#
# Install from the repository root:
#     python -m venv .venv
#     # Windows: .venv\Scripts\activate
#     # macOS/Linux: source .venv/bin/activate
#     python -m pip install --upgrade pip
#     pip install -r requirements.txt
#
# Then run the Gradio UI:
#     cd src/app
#     python main.py
